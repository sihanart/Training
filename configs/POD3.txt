! =====================================================================
! POD 3 - Student
! Mobility Controller (VM)  |  ArubaOS 8.7.1.2
! Controller MGMT : 192.168.10.223
! AP VLAN         : 2603   Subnet: 10.26.3.0/24
! =====================================================================

configure terminal

! ---------- 1) VLAN + SVI (gateway ของ AP VLAN) ----------
vlan 2603
!
interface vlan 2603
 description "POD 3 - Student AP VLAN"
 ip address 10.26.3.1 255.255.255.0
 ip nat inside
 operstate up
!

! ---------- 2) กัน IP ที่ไม่ให้แจก ----------
ip dhcp excluded-address 10.26.3.1 10.26.3.10
ip dhcp excluded-address 10.26.3.241 10.26.3.254

! ---------- 3) DHCP pool ----------
ip dhcp pool POOL-VLAN2603
 network 10.26.3.0 255.255.255.0
 default-router 10.26.3.1
 dns-server 8.8.8.8 1.1.1.1
 domain-name lab.local
 lease 0 8 0
!

! ---------- 4) เปิดบริการ DHCP + บันทึก ----------
service dhcp
write memory
